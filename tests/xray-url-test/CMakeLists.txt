cmake_minimum_required(VERSION 3.24)
project(XrayURLTest LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network)
find_package(Threads REQUIRED)
set(CPR_USE_SYSTEM_CURL ON CACHE BOOL "" FORCE)
set(CPR_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(CPR_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory(../../src/3rdparty/cpr cpr)
add_executable(xray-url-test main.cpp ../../src/gharqad/sys/XrayURLTest.cpp)
target_include_directories(xray-url-test PRIVATE ../../src)
target_link_libraries(xray-url-test PRIVATE Qt6::Core Qt6::Network cpr::cpr Threads::Threads)
if(WIN32)
    target_link_libraries(xray-url-test PRIVATE ws2_32)
endif()
